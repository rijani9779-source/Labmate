<div class="container" id="appContent"></div>

<script>
  // Data Simulasi Awal
  const dataAwal = [
    { id: 101, peminjam: "Ahmad Fauzi", nim: "2210131210005", lab: "Lab Rekayasa Perangkat Lunak", tanggal: "2026-12-01", sesi: "08:00 - 10:00", keperluan: "Praktikum RPL", status: "Disetujui" },
    { id: 102, peminjam: "Siti Aminah", nim: "2210131220014", lab: "Lab Jaringan Komputer", tanggal: "2026-12-02", sesi: "10:00 - 12:00", keperluan: "Simulasi Routing", status: "Menunggu" }
  ];

  let daftarData = JSON.parse(localStorage.getItem("simLabData")) || dataAwal;
  let currentRole = "peminjam";
  let currentMenu = "Beranda";

  const menuConfig = {
    peminjam: ["Beranda", "Jadwal", "Riwayat", "Profil"],
    admin: ["Dashboard", "Jadwal", "Pengajuan", "Profil"]
  };

  function simpanStorage() {
    localStorage.setItem("simLabData", JSON.stringify(daftarData));
  }

  function gantiRole() {
    currentRole = document.getElementById("roleSelect").value;
    currentMenu = menuConfig[currentRole][0];
    renderNav();
    renderHalaman();
  }

  function pindahMenu(menu) {
    currentMenu = menu;
    renderNav();
    renderHalaman();
  }

  function renderNav() {
    const nav = document.getElementById("navMenu");
    nav.innerHTML = menuConfig[currentRole].map(m =>
      `<button class="${m === currentMenu ? 'active' : ''}" onclick="pindahMenu('${m}')">${m}</button>`
    ).join("");
  }

  function renderHalaman() {
    const app = document.getElementById("appContent");

    // 1. POV PEMINJAM: BERANDA
    if (currentRole === "peminjam" && currentMenu === "Beranda") {
      app.innerHTML = `
        <div class="card">
          <h2>Selamat Datang di SIM-LAB</h2>
          <p style="font-size:14px; color:#475569;">Silakan cek ketersediaan jadwal atau ajukan peminjaman ruang laboratorium melalui formulir di bawah ini.</p>
          <form id="formAjukan" onsubmit="prosesAjukan(event)">
            <div class="form-grid">
              <div>
                <label>Pilih Laboratorium</label>
                <select id="lab">
                  <option value="Lab Rekayasa Perangkat Lunak">Lab Rekayasa Perangkat Lunak</option>
                  <option value="Lab Jaringan Komputer">Lab Jaringan Komputer</option>
                  <option value="Lab Multimedia">Lab Multimedia</option>
                </select>
              </div>
              <div>
                <label>Tanggal Peminjaman</label>
                <input type="date" id="tanggal" required />
              </div>
              <div>
                <label>Sesi Waktu</label>
                <select id="sesi">
                  <option value="08:00 - 10:00">Sesi 1 (08:00 - 10:00 WITA)</option>
                  <option value="10:00 - 12:00">Sesi 2 (10:00 - 12:00 WITA)</option>
                  <option value="13:30 - 15:30">Sesi 3 (13:30 - 15:30 WITA)</option>
                </select>
              </div>
              <div>
                <label>Keperluan</label>
                <input type="text" id="keperluan" placeholder="Contoh: Praktikum / Rapat Himakom" required />
              </div>
            </div>
            <button type="submit" class="btn-primary">Ajukan Peminjaman Lab</button>
            <div id="pesanForm" class="alert"></div>
          </form>
        </div>`;
    }

    // 2. POV ADMIN: DASHBOARD
    else if (currentRole === "admin" && currentMenu === "Dashboard") {
      const total = daftarData.length;
      const menunggu = daftarData.filter(d => d.status === "Menunggu").length;
      const disetujui = daftarData.filter(d => d.status === "Disetujui").length;
      app.innerHTML = `
        <div class="stats">
          <div class="stat-card"><h3>3</h3><p>Total Laboratorium Aktif</p></div>
          <div class="stat-card"><h3>${menunggu}</h3><p>Pengajuan Menunggu Verifikasi</p></div>
          <div class="stat-card"><h3>${disetujui}</h3><p>Jadwal Terverifikasi</p></div>
          <div class="stat-card"><h3>${total}</h3><p>Total Transaksi Peminjaman</p></div>
        </div>
        <div class="card">
          <h2>Ringkasan Aktivitas Terbaru</h2>
          <p style="font-size:14px;">Buka menu <strong>Pengajuan</strong> untuk menyetujui atau menolak permohonan peminjaman ruangan yang masuk.</p>
        </div>`;
    }

    // 3. MENU JADWAL (BISA DIAKSES PEMINJAM & ADMIN)
    else if (currentMenu === "Jadwal") {
      const jadwalAktif = daftarData.filter(d => d.status === "Disetujui");
      app.innerHTML = `
        <div class="card">
          <h2>Jadwal Penggunaan Laboratorium (Terverifikasi)</h2>
          <table>
            <thead><tr><th>Laboratorium</th><th>Tanggal</th><th>Sesi</th><th>Penanggung Jawab</th><th>Kegiatan</th></tr></thead>
            <tbody>
              ${jadwalAktif.map(j => `
                <tr>
                  <td><strong>${j.lab}</strong></td>
                  <td>${j.tanggal}</td>
                  <td>${j.sesi}</td>
                  <td>${j.peminjam}</td>
                  <td>${j.keperluan}</td>
                </tr>`).join("")}
            </tbody>
          </table>
        </div>`;
    }

    // 4. POV PEMINJAM: RIWAYAT
    else if (currentRole === "peminjam" && currentMenu === "Riwayat") {
      app.innerHTML = `
        <div class="card">
          <h2>Riwayat Pengajuan Peminjaman Saya</h2>
          <table>
            <thead><tr><th>Laboratorium</th><th>Tanggal & Sesi</th><th>Keperluan</th><th>Status</th></tr></thead>
            <tbody>
              ${daftarData.map(r => `
                <tr>
                  <td><strong>${r.lab}</strong></td>
                  <td>${r.tanggal} (${r.sesi})</td>
                  <td>${r.keperluan}</td>
                  <td><span class="badge ${r.status}">${r.status}</span></td>
                </tr>`).join("")}
            </tbody>
          </table>
        </div>`;
    }

    // 5. POV ADMIN: PENGAJUAN (VERIFIKASI)
    else if (currentRole === "admin" && currentMenu === "Pengajuan") {
      app.innerHTML = `
        <div class="card">
          <h2>Kelola Pengajuan Peminjaman Masuk</h2>
          <table>
            <thead><tr><th>Peminjam</th><th>Laboratorium</th><th>Waktu</th><th>Keperluan</th><th>Status</th><th>Aksi</th></tr></thead>
            <tbody>
              ${daftarData.map((p, idx) => `
                <tr>
                  <td><strong>${p.peminjam}</strong><br><small>${p.nim}</small></td>
                  <td>${p.lab}</td>
                  <td>${p.tanggal}<br><small>${p.sesi}</small></td>
                  <td>${p.keperluan}</td>
                  <td><span class="badge ${p.status}">${p.status}</span></td>
                  <td>
                    ${p.status === "Menunggu" ? `
                      <button class="btn-sm btn-acc" onclick="ubahStatus(${idx}, 'Disetujui')">Setujui</button>
                      <button class="btn-sm btn-rej" onclick="ubahStatus(${idx}, 'Ditolak')">Tolak</button>
                    ` : '-'}
                  </td>
                </tr>`).join("")}
            </tbody>
          </table>
        </div>`;
    }

    // 6. MENU PROFIL (SESUAI POV)
    else if (currentMenu === "Profil") {
      const isAdmin = currentRole === "admin";
      app.innerHTML = `
        <div class="card">
          <h2>Profil Pengguna (${isAdmin ? "Admin Laboratorium" : "Peminjam"})</h2>
          <table>
            <tr><th style="width:200px;">Nama Lengkap</th><td>${isAdmin ? "Drs. Hendra Wijaya, M.Kom." : "Ahmad Fauzi"}</td></tr>
            <tr><th>Nomor Induk (${isAdmin ? "NIP" : "NIM"})</th><td>${isAdmin ? "198504122010121002" : "2210131210005"}</td></tr>
            <tr><th>Peran Sistem</th><td>${isAdmin ? "Administrator Lab" : "Mahasiswa / Peminjam"}</td></tr>
            <tr><th>Program Studi</th><td>Pendidikan Komputer / Ilmu Komputer</td></tr>
          </table>
        </div>`;
    }
  }

  // Logika Validasi Bentrok Jadwal
  function prosesAjukan(e) {
    e.preventDefault();
    const lab = document.getElementById("lab").value;
    const tanggal = document.getElementById("tanggal").value;
    const sesi = document.getElementById("sesi").value;
    const keperluan = document.getElementById("keperluan").value.trim();
    const pesan = document.getElementById("pesanForm");

    const bentrok = daftarData.some(item =>
      item.lab === lab && item.tanggal === tanggal && item.sesi === sesi && item.status !== "Ditolak"
    );

    if (bentrok) {
      pesan.className = "alert error";
      pesan.textContent = "Gagal! Laboratorium sudah dibooking atau sedang diajukan pada jadwal tersebut.";
      return;
    }

    daftarData.push({
      id: Date.now(),
      peminjam: "Ahmad Fauzi",
      nim: "2210131210005",
      lab, tanggal, sesi, keperluan,
      status: "Menunggu"
    });
    simpanStorage();
    pesan.className = "alert success";
    pesan.textContent = "Berhasil! Pengajuan masuk ke daftar Riwayat dan menunggu verifikasi Admin.";
    e.target.reset();
  }

  // Logika Verifikasi Admin
  function ubahStatus(index, statusBaru) {
    daftarData[index].status = statusBaru;
    simpanStorage();
    renderHalaman();
  }

  renderNav();
  renderHalaman();
</script>
