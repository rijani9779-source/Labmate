  // Fungsi Logika Booking
  function mulaiBooking(lab) {
    go("main", { selectedLab: lab, bookingStep: 1 });
  }

  function lanjutDetailBooking() {
    state.tempBooking = {
      nama: document.getElementById("bNama").value || state.user.nama,
      nim: document.getElementById("bNim").value || state.user.id,
      kelas: document.getElementById("bKelas").value || "Kelas A6",
      jurusan: document.getElementById("bJurusan").value || state.user.jurusan
    };
    go("main", { bookingStep: 2 });
  }

  function selesaikanBooking() {
    const keperluan = document.getElementById("bKeperluan").value.trim() || "Praktikum";
    const tanggal = document.getElementById("bTanggal").value;
    const jamMulai = document.getElementById("bMulai").value.trim() || "08.00";
    const jamSelesai = document.getElementById("bSelesai").value.trim() || "10.00";
    const peserta = document.getElementById("bPeserta").value || 25;

    // Cek bentrok jadwal (Lab, Tanggal, dan Jam Mulai sama)
    const isBentrok = bookings.some(b =>
      b.lab === state.selectedLab && b.tanggal === tanggal && b.jamMulai === jamMulai && b.status !== "Ditolak"
    );

    if (isBentrok) {
      go("main", { bookingStep: 4 }); // Layar Merah Gagal
      return;
    }

    bookings.unshift({
      id: Date.now(),
      ...state.tempBooking,
      lab: state.selectedLab,
      tanggal, jamMulai, jamSelesai, peserta, keperluan,
      status: "Menunggu"
    });
    saveBookings();
    go("main", { bookingStep: 3 }); // Layar Hijau Berhasil
  }

  // Fungsi Verifikasi Admin
  function bukaDetailAdmin(id) {
    const item = bookings.find(b => b.id === id);
    go("main", { selectedAdminItem: item });
  }

  function verifikasiPengajuan(id, statusBaru) {
    bookings = bookings.map(b => b.id === id ? { ...b, status: statusBaru } : b);
    saveBookings();
    go("main", { selectedAdminItem: null });
  }
